Script Coletor de Curtidas de Fotos (Navegador)

Abra o post no Instagram Web, clique nas curtidas e cole este script no Console (F12).

(async function coletarCurtidas() {
  const modal = document.querySelector('div[role="dialog"]');
  if (!modal) {
    console.error("Abra a janela de curtidas do post antes de rodar o script.");
    return;
  }

  const scrollContainer = Array.from(modal.querySelectorAll('div')).find(
    el => el.scrollHeight > el.clientHeight && el.clientHeight > 100
  );

  if (!scrollContainer) {
    console.error("Container de rolagem não encontrado.");
    return;
  }

  const usuarios = new Map();
  let tentativasSemNovos = 0;

  console.log("Iniciando coleta de curtidas...");

  while (tentativasSemNovos < 5) {
    const tamanhoAnterior = usuarios.size;
    const itens = modal.querySelectorAll('a[role="link"]');

    itens.forEach(link => {
      const href = link.getAttribute('href');
      if (href && href.startsWith('/') && href.split('/').length === 3) {
        const username = href.replace(/\//g, '');
        const containerLinha = link.closest('div[style]') || link.parentElement?.parentElement;
        const img = containerLinha ? containerLinha.querySelector('img') : null;
        const altText = img ? (img.getAttribute('alt') || '') : '';
        const src = img ? (img.getAttribute('src') || '') : '';

        const semFoto = !src || src.includes('44884218_345707102882519_2446069589734326272_n') || altText.includes('Foto do perfil padrão');

        if (!usuarios.has(username)) {
          usuarios.set(username, {
            username: username,
            sem_foto_perfil: semFoto,
            coletado_em: new Date().toISOString()
          });
        }
      }
    });

    if (usuarios.size === tamanhoAnterior) {
      tentativasSemNovos++;
    } else {
      tentativasSemNovos = 0;
      console.log(`Perfis coletados até agora: ${usuarios.size}`);
    }

    scrollContainer.scrollTop += 600;
    await new Promise(r => setTimeout(r, 1500));
  }

  const resultado = Array.from(usuarios.values());
  const blob = new Blob([JSON.stringify(resultado, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'curtidas_coletadas.json';
  a.click();

  console.log(`Concluído! ${resultado.length} perfis baixados em 'curtidas_coletadas.json'.`);
})();
Gera o arquivo JSON pronto para carregar no aplicativo.
